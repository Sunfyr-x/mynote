## [[数据结构.pdf#page=180&selection=76,0,76,6|树的存储结构]]

- **双亲表示法**：孩子有个指向双亲的指针（牛魔的就一个结点还要叫双亲）
- **孩子表示法**：给每个结点设置一个孩子列表，所有孩子列表的头指针设置在顺序表中
- **孩子兄弟表示法**：左孩子右兄弟，中间存数据，可非常自然地映射为一颗二叉树，复用二叉树的算法

```c
/************双亲表示法************/

#define MAX_TREE_SIZE = 100

//定义结点
typedef struct{
	//数据部分
	ElemType data;
	//伪指针部分
	int parent;
}PTNode;

//定义树
typedef struct{
	PTNode nodes[MAX_TREE_SIZE];
	int n;
}PTree;

/************孩子表示法************/

#define MAX_TREE_SIZE = 100

//定义结点
typedef struct CNode{
	//数据部分
	ElemType data;
	//指针部分
	struct CNode *next;
}CNode

typedef struct CTree{
	CNode nodes[MAX_TREE_SIZE];
	int n;
}

/**********孩子兄弟表示法**********/

typedef struct CSNode{
	//数据
	ElemType data;
	//左孩子右兄弟
	struct CSNode *firstchild, *nextsibling;
}CSNode, *CSTree;
```
