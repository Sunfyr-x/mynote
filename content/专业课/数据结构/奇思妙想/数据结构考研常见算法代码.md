## 1. 线性表

### 1.1 顺序表

```c
#define MAXSIZE 100
typedef struct {
    int data[MAXSIZE];
    int length;
} SqList;

// 按值查找，返回下标；找不到返回 -1。O(n)
int LocateElem(SqList L, int e) {
    for (int i = 0; i < L.length; i++)
        if (L.data[i] == e) return i;
    return -1;
}

// 在第 i 个位置插入 e（i 为逻辑序号，从 1 开始）。O(n)
int ListInsert(SqList *L, int i, int e) {
    if (i < 1 || i > L->length + 1 || L->length == MAXSIZE) return 0;
    for (int j = L->length - 1; j >= i - 1; j--)   // 元素后移，从后往前
        L->data[j + 1] = L->data[j];
    L->data[i - 1] = e;
    L->length++;
    return 1;
}

// 删除第 i 个位置的元素，用 e 带回。O(n)
int ListDelete(SqList *L, int i, int *e) {
    if (i < 1 || i > L->length) return 0;
    *e = L->data[i - 1];
    for (int j = i; j < L->length; j++)            // 元素前移
        L->data[j - 1] = L->data[j];
    L->length--;
    return 1;
}
```

**易错点**：插入/删除时边界判断 `i` 的范围是 `[1, length+1]` / `[1, length]`；移动方向别写反。

### 1.2 单链表

```c
typedef struct LNode {
    int data;
    struct LNode *next;
} LNode, *LinkList;

// 头插法建表（结果逆序）。O(n)
LinkList CreateHead(int a[], int n) {
    LinkList L = (LinkList)malloc(sizeof(LNode));
    L->next = NULL;
    for (int i = 0; i < n; i++) {
        LNode *p = (LNode*)malloc(sizeof(LNode));
        p->data = a[i];
        p->next = L->next;      // 新结点插到头结点之后
        L->next = p;
    }
    return L;
}

// 尾插法建表（需尾指针 r 保存表尾，否则每次遍历找尾 O(n^2)）。O(n)
LinkList CreateTail(int a[], int n) {
    LinkList L = (LinkList)malloc(sizeof(LNode));
    L->next = NULL;
    LNode *r = L;               // r 始终指向表尾
    for (int i = 0; i < n; i++) {
        LNode *p = (LNode*)malloc(sizeof(LNode));
        p->data = a[i];
        r->next = p;
        r = p;
    }
    r->next = NULL;
    return L;
}

// 按序号查找第 i 个结点（i 从 1 开始）。O(n)
LNode* GetElem(LinkList L, int i) {
    if (i < 0) return NULL;      // i=0 返回头结点
    LNode *p = L;
    int j = 0;
    while (p && j < i) { p = p->next; j++; }
    return p;
}

// 在 p 所指结点后插入 s。O(1)（若无前驱则需从头找，退化 O(n)）
void InsertAfter(LNode *p, LNode *s) {
    s->next = p->next;
    p->next = s;
}

// 删除 p 的后继结点。O(1)
int DeleteAfter(LNode *p, int *e) {
    if (!p || !p->next) return 0;
    LNode *q = p->next;
    *e = q->data;
    p->next = q->next;
    free(q);
    return 1;
}
```

### 1.5 高频算法题

```c
// 逆置单链表（头插法，三指针）。O(n)，O(1)
LinkList Reverse(LinkList L) {
    LNode *p = L->next, *r;
    L->next = NULL;              // 头结点摘下
    while (p) {
        r = p->next;             // 保存后继
        p->next = L->next;       // 头插
        L->next = p;
        p = r;
    }
    return L;
}

// 找链表中倒数第 k 个结点（双指针）。O(n)
LNode* FindKthFromEnd(LinkList L, int k) {
    LNode *fast = L, *slow = L;
    for (int i = 0; i < k && fast; i++) fast = fast->next;
    if (!fast) return NULL;      // k 大于表长
    while (fast) { fast = fast->next; slow = slow->next; }
    return slow;
}

// 判断链表是否有环（快慢指针）。O(n)
int HasCycle(LinkList L) {
    LNode *fast = L, *slow = L;
    while (fast && fast->next) {
        fast = fast->next->next;
        slow = slow->next;
        if (fast == slow) return 1;
    }
    return 0;
}

// 删除有序链表中所有值为 x 的结点。O(n)
void DeleteX(LinkList L, int x) {
    LNode *pre = L, *p = L->next;
    while (p) {
        if (p->data == x) {
            pre->next = p->next;
            free(p);
            p = pre->next;
        } else {
            pre = p; p = p->next;
        }
    }
}

// 合并两个有序链表（归并，非递减）。O(n+m)
LinkList Merge(LinkList A, LinkList B) {
    LNode *p = A->next, *q = B->next, *r;
    LinkList C = A;              // 用 A 的头结点作 C 的头
    C->next = NULL;
    r = C;
    while (p && q) {
        if (p->data <= q->data) { r->next = p; p = p->next; }
        else { r->next = q; q = q->next; }
        r = r->next;
    }
    r->next = p ? p : q;         // 剩余部分直接挂上
    free(B);                     // B 的头结点没用，释放
    return C;
}
```

## 2. 栈与队列

### 2.1 顺序栈

```c
typedef struct {
    int data[MAXSIZE];
    int top;            // 栈顶指针，指向下一个可插入位置；空栈时为 0
} SqStack;

void InitStack(SqStack *S) { S->top = 0; }
int StackEmpty(SqStack S) { return S.top == 0; }
int Push(SqStack *S, int e) {
    if (S->top == MAXSIZE) return 0;
    S->data[S->top++] = e;
    return 1;
}
int Pop(SqStack *S, int *e) {
    if (StackEmpty(*S)) return 0;
    *e = S->data[--S->top];
    return 1;
}
int GetTop(SqStack S, int *e) {
    if (StackEmpty(S)) return 0;
    *e = S->data[S.top - 1];
    return 1;
}
```

### 2.2 链栈

```c
typedef struct StackNode {

// 入栈：头插。O(1)
void Push(LinkStack *S, int e) {

// 出栈。O(1)
int Pop(LinkStack *S, int *e) {
    if (!(*S)) return 0;
```

### 2.3 循环队列（顺序）

```c
typedef struct {
    int data[MAXSIZE];
    int front, rear;    // front 指向队头元素，rear 指向下一个空位
} SqQueue;

void InitQueue(SqQueue *Q) { Q->front = Q->rear = 0; }
int QueueEmpty(SqQueue Q) { return Q.front == Q.rear; }
// 牺牲一个单元区分队空与队满：队满条件 (rear+1)%MAX == front
int EnQueue(SqQueue *Q, int e) {
    if ((Q->rear + 1) % MAXSIZE == Q->front) return 0;
    Q->data[Q->rear] = e;
    Q->rear = (Q->rear + 1) % MAXSIZE;
    return 1;
}
int DeQueue(SqQueue *Q, int *e) {
    if (QueueEmpty(*Q)) return 0;
    *e = Q->data[Q->front];
    Q->front = (Q->front + 1) % MAXSIZE;
    return 1;
}
// 队长：(rear - front + MAXSIZE) % MAXSIZE
```

### 2.4 链队列

```c
typedef struct QNode { int data; struct QNode *next; } QNode;
typedef struct { QNode *front, *rear; } LinkQueue;

void InitQueue(LinkQueue *Q) {
    Q->front = Q->rear = (QNode*)malloc(sizeof(QNode)); // 头结点
    Q->front->next = NULL;
}
int EnQueue(LinkQueue *Q, int e) {
    QNode *p = (QNode*)malloc(sizeof(QNode));
    p->data = e; p->next = NULL;
    Q->rear->next = p;
    Q->rear = p;
    return 1;
}
int DeQueue(LinkQueue *Q, int *e) {
    if (Q->front == Q->rear) return 0;
    QNode *p = Q->front->next;
    *e = p->data;
    Q->front->next = p->next;
    if (Q->rear == p) Q->rear = Q->front;   // 若只剩一个元素，更新 rear
    free(p);
    return 1;
}
```

```c
// 括号匹配：仅处理 () [] {}
int BracketCheck(char *str) {
    SqStack S; InitStack(&S);
    for (int i = 0; str[i]; i++) {
        char c = str[i];
        if (c == '(' || c == '[' || c == '{') Push(&S, c);
```

## 3. 串与 KMP

### 3.1 BF（朴素匹配）

```c
// 返回主串 S 中模式 T 首次出现的位置（下标），失败返回 -1。O(nm)
int BF(char *S, char *T) {
    int i = 0, j = 0;
    int n = strlen(S), m = strlen(T);
    while (i < n && j < m) {
        if (S[i] == T[j]) { i++; j++; }
        else { i = i - j + 1; j = 0; }   // 主串回溯
    }
    return j == m ? i - m : -1;
}
```

### 3.2 KMP

```c
// 求 next 数组：next[j] 表示模式串 T[0..j-1] 的最长相等前后缀长度
void GetNext(char *T, int next[]) {
    int j = 0, k = -1;     // k 即 next[j]
    next[0] = -1;          // 约定：next[0] = -1
    int m = strlen(T);
    while (j < m - 1) {
        if (k == -1 || T[j] == T[k]) {
            j++; k++;
            // 若优化（nextval）：当 T[j]==T[k] 时，next[j] 直接等于 next[k]
            next[j] = k;
        } else {
            k = next[k];   // 回溯
        }
    }
}

// KMP 匹配。O(n+m)
int KMP(char *S, char *T, int next[]) {
    int i = 0, j = 0;
    int n = strlen(S), m = strlen(T);
    while (i < n && j < m) {
        if (j == -1 || S[i] == T[j]) { i++; j++; }
        else j = next[j];      // 模式串指针回溯，主串 i 不回溯
    }
    return j == m ? i - m : -1;
}
```

**手算 next 口诀**：`next[0]=-1`，`next[1]=0`；对 `j>=2`，看 `T[0..j-1]` 的最长相等前后缀长度。
**易错点**：`next[0]` 到底是 `-1` 还是 `0` 取决于教材约定，考试按题目给的定义写。

### 4.4 由遍历序列建树

```c
// 由先序 + 中序建树。pre[preL..preR], in[inL..inR]
// 前提：无重复值。O(n)
BiTree CreateTree(int pre[], int in[], int preL, int preR, int inL, int inR) {
    if (preL > preR) return NULL;
    BiTree root = (BiTree)malloc(sizeof(BiTNode));
    root->data = pre[preL];
    root->lchild = root->rchild = NULL;
    int k;
    for (k = inL; k <= inR; k++)
        if (in[k] == pre[preL]) break;
    int leftLen = k - inL;
    root->lchild = CreateTree(pre, in, preL + 1, preL + leftLen, inL, k - 1);
    root->rchild = CreateTree(pre, in, preL + leftLen + 1, preR, k + 1, inR);
    return root;
}
// 由后序 + 中序建树：后序末位是根，同理
```

```c
int IsBalanced(BiTree T, int *height) {
    if (!T) { *height = 0; return 1; }
    int lh, rh;
    if (!IsBalanced(T->lchild, &lh) || !IsBalanced(T->rchild, &rh)) return 0;
    *height = (lh > rh ? lh : rh) + 1;
    return abs(lh - rh) <= 1;
}
```

### 4.7 哈夫曼树

```c
typedef struct {
    int weight;
    int parent, lchild, rchild;
} HTNode, *HuffmanTree;

// 在 [1..n] 中选两个 parent==0 且权值最小的结点
void SelectMin(HuffmanTree HT, int n, int *s1, int *s2) {
    int i, min1 = 0x3f3f3f3f, min2 = 0x3f3f3f3f;
    *s1 = *s2 = 0;
    for (i = 1; i <= n; i++) {
        if (HT[i].parent) continue;
        if (HT[i].weight < min1) { min2 = min1; *s2 = *s1; min1 = HT[i].weight; *s1 = i; }
        else if (HT[i].weight < min2) { min2 = HT[i].weight; *s2 = i; }
    }
}
// 建树：HT 空间为 2n，前 n 个为叶子。O(n^2)（可用堆优化到 O(n log n)）
void CreateHuffmanTree(HuffmanTree *HT, int w[], int n) {
    int m = 2 * n - 1;
    *HT = (HTNode*)malloc((m + 1) * sizeof(HTNode));   // 0 号不用
    for (int i = 1; i <= n; i++) (*HT)[i] = (HTNode){w[i-1], 0, 0, 0};
    for (int i = n + 1; i <= m; i++) (*HT)[i] = (HTNode){0, 0, 0, 0};
    for (int i = n + 1; i <= m; i++) {
        int s1, s2;
        SelectMin(*HT, i - 1, &s1, &s2);
        (*HT)[i].weight = (*HT)[s1].weight + (*HT)[s2].weight;
        (*HT)[i].lchild = s1; (*HT)[i].rchild = s2;
        (*HT)[s1].parent = (*HT)[s2].parent = i;
    }
}
// WPL = 所有叶子权值 × 路径长度之和。O(n)
int WPL(HuffmanTree HT, int n) {
    int wpl = 0;
    for (int i = 1; i <= n; i++) {
        int len = 0, p = HT[i].parent;
        while (p) { len++; p = HT[p].parent; }
        wpl += HT[i].weight * len;
    }
    return wpl;
}
```

### 4.8 并查集（树的应用）

```c
int parent[MAXSIZE];
void InitSet(int n) { for (int i = 0; i < n; i++) parent[i] = -1; }
int Find(int x) { return parent[x] < 0 ? x : (parent[x] = Find(parent[x])); } // 路径压缩
void Union(int x, int y) {      // 按规模合并
    int rx = Find(x), ry = Find(y);
    if (rx == ry) return;
    if (parent[rx] > parent[ry]) { parent[ry] += parent[rx]; parent[rx] = ry; }
    else { parent[rx] += parent[ry]; parent[ry] = rx; }
}
```

### 5.2 BFS 与 DFS

```c
int visited[MAXSIZE];

// 邻接矩阵 BFS。O(V^2)
void BFS(MGraph G, int v) {
    int q[MAXSIZE], front = 0, rear = 0;
    printf("%d ", v); visited[v] = 1; q[rear++] = v;
    while (front < rear) {
        int x = q[front++];
        for (int i = 0; i < G.vexnum; i++)
            if (G.arcs[x][i] && !visited[i]) {
                printf("%d ", i); visited[i] = 1; q[rear++] = i;
            }
    }
}
```

```c
#define INF 0x3f3f3f3f

// Prim：从 u 出发。O(V^2)，适合稠密图
void Prim(MGraph G, int u) {
    int lowcost[MAXSIZE], closest[MAXSIZE];
    for (int i = 0; i < G.vexnum; i++) {
        lowcost[i] = G.arcs[u][i];
        closest[i] = u;
    }
    lowcost[u] = 0;
    for (int i = 1; i < G.vexnum; i++) {
        int min = INF, k = -1;
        for (int j = 0; j < G.vexnum; j++)
            if (lowcost[j] && lowcost[j] < min) { min = lowcost[j]; k = j; }
        lowcost[k] = 0;                  // 加入 U 集
        for (int j = 0; j < G.vexnum; j++)
            if (G.arcs[k][j] && G.arcs[k][j] < lowcost[j]) {
                lowcost[j] = G.arcs[k][j];
                closest[j] = k;
            }
    }
}

// Kruskal：边按权排序 + 并查集。O(E log E)，适合稀疏图
typedef struct { int u, v, w; } Edge;
int cmp(const void *a, const void *b) { return ((Edge*)a)->w - ((Edge*)b)->w; }
void Kruskal(ALGraph G, Edge e[], int n) {
    qsort(e, G.arcnum, sizeof(Edge), cmp);
    InitSet(G.vexnum);
    int count = 0;
    for (int i = 0; i < G.arcnum && count < G.vexnum - 1; i++) {
        if (Find(e[i].u) != Find(e[i].v)) {
            Union(e[i].u, e[i].v);
            count++;
        }
    }
}
```

### 5.4 最短路径

```c
// Dijkstra（无负权）。O(V^2)
void Dijkstra(MGraph G, int v) {
    int dist[MAXSIZE], path[MAXSIZE], s[MAXSIZE];
    for (int i = 0; i < G.vexnum; i++) {
        dist[i] = G.arcs[v][i];
        s[i] = 0; path[i] = dist[i] < INF ? v : -1;
    }
    s[v] = 1; dist[v] = 0;
    for (int i = 1; i < G.vexnum; i++) {
        int min = INF, u = -1;
        for (int j = 0; j < G.vexnum; j++)
            if (!s[j] && dist[j] < min) { min = dist[j]; u = j; }
        if (u == -1) break;
        s[u] = 1;
        for (int j = 0; j < G.vexnum; j++)
            if (!s[j] && G.arcs[u][j] && dist[u] + G.arcs[u][j] < dist[j]) {
                dist[j] = dist[u] + G.arcs[u][j];
                path[j] = u;
            }
    }
}

// Floyd（全源最短路径）。O(V^3)
void Floyd(MGraph G, int dist[][MAXSIZE]) {
    for (int i = 0; i < G.vexnum; i++)
        for (int j = 0; j < G.vexnum; j++)
            dist[i][j] = G.arcs[i][j];
    for (int k = 0; k < G.vexnum; k++)
        for (int i = 0; i < G.vexnum; i++)
            for (int j = 0; j < G.vexnum; j++)
                if (dist[i][k] < INF && dist[k][j] < INF &&
                    dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
}

// Bellman-Ford：可判负权环。O(VE)
int BellmanFord(Edge e[], int n, int src, int dist[]) {
    for (int i = 0; i < n; i++) dist[i] = INF;
    dist[src] = 0;
    for (int i = 1; i < n; i++)
        for (int j = 0; j < n; j++)
            if (dist[e[j].u] < INF && dist[e[j].u] + e[j].w < dist[e[j].v])
                dist[e[j].v] = dist[e[j].u] + e[j].w;
    for (int j = 0; j < n; j++)      // 再松弛一次，还能更新则说明有负权环
        if (dist[e[j].u] < INF && dist[e[j].u] + e[j].w < dist[e[j].v)
            return 0;
    return 1;
}
```

```c
// 拓扑排序（基于入度，邻接表）。O(V+E)
int TopoSort(ALGraph G, int topo[]) {
    int indeg[MAXSIZE] = {0}, q[MAXSIZE], front = 0, rear = 0, cnt = 0;
    for (int i = 0; i < G.vexnum; i++)
        for (ArcNode *p = G.adjlist[i].first; p; p = p->next)
            indeg[p->adjvex]++;
    for (int i = 0; i < G.vexnum; i++)
        if (!indeg[i]) q[rear++] = i;
    while (front < rear) {
        int v = q[front++]; topo[cnt++] = v;
        for (ArcNode *p = G.adjlist[v].first; p; p = p->next)
            if (--indeg[p->adjvex] == 0) q[rear++] = p->adjvex;
    }
    return cnt == G.vexnum;     // 0 表示有环
}
```

```c
// 顺序查找（哨兵法）。O(n)
int SeqSearch(int a[], int n, int key) {
    a[0] = key;                 // 0 号单元作哨兵
    int i = n;
    while (a[i] != key) i--;
    return i;                    // 返回 0 表示失败
}

// 折半查找（要求有序）。O(log n)
int BinarySearch(int a[], int n, int key) {
    int low = 0, high = n - 1, mid;
    while (low <= high) {
        mid = low + (high - low) / 2;   // 防溢出
        if (a[mid] == key) return mid;
        else if (a[mid] < key) low = mid + 1;
        else high = mid - 1;
    }
    return -1;
}

// 折半查找判定树：成功 ASL = (1×1 + 2×2 + ... + h×2^{h-1})/n，近似 log(n+1)-1
```

### 6.2 二叉排序树（BST）

```c
typedef struct BSTNode {
    int data;
    struct BSTNode *lchild, *rchild;
} BSTNode, *BSTree;

// 查找。O(h)
BSTree BSTSearch(BSTree T, int key) {
    if (!T || T->data == key) return T;
    if (key < T->data) return BSTSearch(T->lchild, key);
    else return BSTSearch(T->rchild, key);
}

// 插入。O(h)
int BSTInsert(BSTree *T, int key) {
    if (!(*T)) {
        *T = (BSTree)malloc(sizeof(BSTNode));
        (*T)->data = key; (*T)->lchild = (*T)->rchild = NULL;
        return 1;
    }
    if (key == (*T)->data) return 0;
    if (key < (*T)->data) return BSTInsert(&(*T)->lchild, key);
    else return BSTInsert(&(*T)->rchild, key);
}

// 删除。O(h) —— 分三种情况
int BSTDelete(BSTree *T, int key) {
    if (!(*T)) return 0;
    if (key < (*T)->data) return BSTDelete(&(*T)->lchild, key);
    if (key > (*T)->data) return BSTDelete(&(*T)->rchild, key);
    // 找到待删结点 *T
    BSTree q, s;
    if (!(*T)->lchild) { q = *T; *T = (*T)->rchild; free(q); }       // 情况1：无左孩子
    else if (!(*T)->rchild) { q = *T; *T = (*T)->lchild; free(q); }  // 情况2：无右孩子
    else {                                                           // 情况3：双孩子
        s = (*T)->lchild;
        while (s->rchild) s = s->rchild;    // 找左子树最右（前驱）
        (*T)->data = s->data;               // 用前驱值覆盖
        BSTDelete(&(*T)->lchild, s->data);  // 转删前驱
    }
    return 1;
}
// 平均查找长度 ASL：与树高有关，最坏 O(n)（退化为链表），平衡二叉树可降到 O(log n)
```

```c
// 直接插入排序。O(n^2)，稳定，适合基本有序/小规模
void InsertSort(int a[], int n) {
    for (int i = 1; i < n; i++) {
        int tmp = a[i], j = i - 1;
        while (j >= 0 && a[j] > tmp) { a[j + 1] = a[j]; j--; }
        a[j + 1] = tmp;
    }
}

// 折半插入排序：查找插入位置用折半，移动次数不变。O(n^2)
// 希尔排序：按增量 dk 分组插入，dk 逐次缩小至 1。O(n^1.3~n^2)，不稳定
void ShellSort(int a[], int n) {
    for (int dk = n / 2; dk >= 1; dk /= 2)
        for (int i = dk; i < n; i++) {
            int tmp = a[i], j = i - dk;
            while (j >= 0 && a[j] > tmp) { a[j + dk] = a[j]; j -= dk; }
            a[j + dk] = tmp;
        }
}
```

```c
// 冒泡排序。O(n^2)，稳定
void BubbleSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (a[j] > a[j + 1]) { int t = a[j]; a[j] = a[j+1]; a[j+1] = t; }
}

// 快速排序。平均 O(n log n)，最坏 O(n^2)，不稳定
int Partition(int a[], int low, int high) {
    int pivot = a[low];              // 选首元素作枢轴
    while (low < high) {
        while (low < high && a[high] >= pivot) high--;
        a[low] = a[high];            // 比枢轴小的移到左端
        while (low < high && a[low] <= pivot) low++;
        a[high] = a[low];            // 比枢轴大的移到右端
    }
    a[low] = pivot;
    return low;
}
void QuickSort(int a[], int low, int high) {
    if (low < high) {
        int p = Partition(a, low, high);
        QuickSort(a, low, p - 1);
        QuickSort(a, p + 1, high);
    }
}
```

```c
// 简单选择排序。O(n^2)，不稳定
void SelectSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int k = i;
        for (int j = i + 1; j < n; j++)
            if (a[j] < a[k]) k = j;
        if (k != i) { int t = a[i]; a[i] = a[k]; a[k] = t; }
    }

// 堆排序。O(n log n)，不稳定
void HeapAdjust(int a[], int k, int n) {   // 将以 k 为根的子堆调整为大根堆
    int tmp = a[k];
    for (int i = 2 * k + 1; i < n; i = 2 * i + 1) {  // 数组从 0 开始，左孩子为 2k+1
        if (i + 1 < n && a[i] < a[i + 1]) i++;         // 选较大的孩子
        if (tmp >= a[i]) break;
        a[k] = a[i]; k = i;
    }
    a[k] = tmp;
}
void HeapSort(int a[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) HeapAdjust(a, i, n);  // 建堆 O(n)
    for (int i = n - 1; i > 0; i--) {
        int t = a[0]; a[0] = a[i]; a[i] = t;                    // 堆顶与末尾交换
        HeapAdjust(a, 0, i);
    }
}
// 注：若数组下标从 1 开始，左孩子为 2k，右孩子为 2k+1，建堆从 n/2 开始
```

```c
// 二路归并（递归）。O(n log n)，稳定，需 O(n) 辅助空间
int *B;      // 辅助数组
void Merge(int a[], int low, int mid, int high) {
    for (int i = low; i <= high; i++) B[i] = a[i];
    int i = low, j = mid + 1, k = low;
    while (i <= mid && j <= high) {
        if (B[i] <= B[j]) a[k++] = B[i++];
        else a[k++] = B[j++];
    }
    while (i <= mid) a[k++] = B[i++];
    while (j <= high) a[k++] = B[j++];
}
void MergeSort(int a[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;
        MergeSort(a, low, mid);
        MergeSort(a, mid + 1, high);
        Merge(a, low, mid, high);
    }
}
```

### 8.1 散列函数与冲突处理

```c
#define P 13
typedef struct {
    int key;
    int info;       // 0 空，1 有数据，2 已删除（用于开放定址的懒惰删除）
} HashTable;

// 除留余数法
int Hash(int key) { return key % P; }

// 线性探测：hi = (H(key) + i) % m，i=1,2,...,m-1
int LinearProbe(HashTable HT[], int key) {
    int i = Hash(key), start = i, cnt = 0;
    while (HT[i].info == 1 && HT[i].key != key) {
        i = (i + 1) % P;
        if (++cnt == P) return -1;     // 表满
        if (i == start) return -1;
    }
    return i;
}
// 平方探测：hi = (H(key) ± i²) % m，需表长取 4k+3 的素数才能保证探到全表
// 双散列：hi = (H1(key) + i·H2(key)) % m，H2(key) 与 m 互素
// 链地址法：每个桶挂一个单链表，冲突元素直接尾插
```

## 9. 综合应用题套路

### 9.1 递归 → 非递归
三件套：**栈 + 循环 + 状态标记**。先写出递归定义，再用手动栈模拟函数调用（如二叉树后序、汉诺塔）。

### 9.2 求路径 / 判连通 / 拓扑
- 图的路径问题：BFS 求无权图最短路径（记录前驱数组 `path[]`）
- 判环：拓扑排序（有向图） / DFS 三色标记（无向图）
- 连通分量：DFS/BFS 计数

### 9.3 树的相关结论（常考填空）
- 二叉树第 i 层最多 2^(i-1) 个结点；深度为 h 的二叉树最多 2^h - 1 个结点
- 叶子数 n0 = n2 + 1（度为 2 的结点数 + 1）
- 完全二叉树：n 个结点，树高 h = ⌊log₂n⌋ + 1；双亲 i 的孩子为 2i、2i+1
- 哈夫曼树没有度为 1 的结点；n 个叶子对应 2n-1 个结点

### 9.4 时间复杂度求法
- 单层循环：`for(i=1;i<n;i*=2)` → O(log n)
- 双重循环：内外独立则相乘，内循环随外循环变化则逐项相加
- 递归：写递推式 T(n)=aT(n/b)+f(n)，用主定理
